# Agent API Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Embed an HTTP server in Colditz Escape so an AI agent can screenshot the game, read state, and inject input via curl.

**Architecture:** New `agent_api.c/.h` service a non-blocking localhost socket from a recurring `glutTimerFunc` tick; frames cached by `glReadPixels` in `glut_display`; input written to the game's own `key_down[]` arrays. No threads.

**Tech Stack:** C (C99), POSIX sockets, vendored `stb_image_write.h` for PNG, shell + curl + python3 for tests.

## Global Constraints

- Bind to 127.0.0.1 only. Default port 8765, `-a <port>` overrides.
- API off unless `-a` given. Zero overhead when off (no glReadPixels).
- Never block the GLUT loop: listener non-blocking; accepted connection uses 200 ms SO_RCVTIMEO/SO_SNDTIMEO; at most one request serviced per tick.
- Request cap 4 KB. Responses: `Connection: close`.
- All new code `#if !defined(WIN32) && !defined(PSP)` guarded.
- Repo: `/Users/fredriksafsten/games/colditz-escape/src`, branch `agent-api`.
- Test data: run game binary from `../Colditz Escape/` (data files live there).

```c
#if !defined(WIN32) && !defined(PSP)
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <sys/time.h>
#include "agent_api.h"

bool agent_api_enabled = false;
static int listen_fd = -1;

void agent_api_init(uint16_t port)
{
    struct sockaddr_in addr;
    int one = 1;
    listen_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (listen_fd < 0) { perror("agent_api: socket"); return; }
    setsockopt(listen_fd, SOL_SOCKET, SO_REUSEADDR, &one, sizeof(one));
    memset(&addr, 0, sizeof(addr));
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
    addr.sin_port = htons(port);
    if (bind(listen_fd, (struct sockaddr*)&addr, sizeof(addr)) < 0 ||
        listen(listen_fd, 4) < 0)
    {
        perror("agent_api: bind/listen (API disabled)");
        close(listen_fd);
        listen_fd = -1;
        return;
    }
    fcntl(listen_fd, F_SETFL, O_NONBLOCK);
    agent_api_enabled = true;
    printf("agent_api: listening on 127.0.0.1:%u\n", port);
}

static void send_response(int fd, int code, const char* ctype,
                          const void* body, size_t len)
{
    char hdr[256];
    const char* msg = (code==200)?"OK":(code==400)?"Bad Request":
                      (code==404)?"Not Found":"Error";
    int n = snprintf(hdr, sizeof(hdr),
        "HTTP/1.0 %d %s\r\nContent-Type: %s\r\n"
        "Content-Length: %zu\r\nConnection: close\r\n\r\n",
        code, msg, ctype, len);
    write(fd, hdr, n);
    size_t off = 0;
    while (off < len) {
        ssize_t w = write(fd, (const char*)body + off, len - off);
        if (w <= 0) break;
        off += (size_t)w;
    }
}

static void handle_state(int fd)
{
    const char* json = "{\"ok\":true}";
    send_response(fd, 200, "application/json", json, strlen(json));
}

static void handle_request(int fd)
{
    static char req_buf[4096];
    char method[8] = "", path[64] = "";
    ssize_t n = recv(fd, req_buf, sizeof(req_buf) - 1, 0);
    if (n <= 0) return;
    req_buf[n] = '\0';
    if (sscanf(req_buf, "%7s %63s", method, path) != 2) {
        send_response(fd, 400, "text/plain", "bad request", 11);
        return;
    }
    if (!strcmp(method, "GET") && !strcmp(path, "/state"))
        handle_state(fd);
    else
        send_response(fd, 404, "text/plain", "not found", 9);
}

void agent_api_tick(void)
{
    struct timeval tv = { 0, 200000 };  // 200 ms cap per request
    int fd;
    if (listen_fd < 0) return;
    fd = accept(listen_fd, NULL, NULL);
    if (fd < 0) return;
    setsockopt(fd, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
    setsockopt(fd, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));
    handle_request(fd);
    close(fd);
}
#endif
```

```c
#define STB_IMAGE_WRITE_IMPLEMENTATION
#define STBI_WRITE_NO_STDIO
#include "stb_image_write.h"
#if defined(__APPLE__)
#include <OpenGL/gl.h>
#else
#include <GL/gl.h>
#endif

static uint8_t* frame_buf = NULL;
static int frame_w = 0, frame_h = 0;

void agent_api_capture(void)
{
    if (!agent_api_enabled) return;
    if (frame_w != gl_width || frame_h != gl_height) {
        free(frame_buf);
        frame_buf = malloc((size_t)gl_width * gl_height * 3);
        frame_w = gl_width; frame_h = gl_height;
    }
    if (!frame_buf) return;
    glPixelStorei(GL_PACK_ALIGNMENT, 1);
    glReadPixels(0, 0, frame_w, frame_h, GL_RGB, GL_UNSIGNED_BYTE, frame_buf);
}

static void png_append(void* ctx, void* data, int size)
{
    /* grow-buffer callback for stb */
    struct growbuf { uint8_t* p; size_t len, cap; } *g = ctx;
    if (g->len + size > g->cap) {
        g->cap = (g->len + size) * 2;
        g->p = realloc(g->p, g->cap);
    }
    memcpy(g->p + g->len, data, size);
    g->len += size;
}

static void handle_screen(int fd)
{
    struct growbuf { uint8_t* p; size_t len, cap; } g = { NULL, 0, 0 };
    if (!frame_buf) {
        send_response(fd, 404, "text/plain", "no frame yet", 12);
        return;
    }
    /* GL rows are bottom-up: point stb at the last row, negative stride */
    stbi_write_png_to_func(png_append, &g, frame_w, frame_h, 3,
                           frame_buf + (size_t)(frame_h-1)*frame_w*3,
                           -frame_w*3);
    if (g.p) {
        send_response(fd, 200, "image/png", g.p, g.len);
        free(g.p);
    } else
        send_response(fd, 500, "text/plain", "encode failed", 13);
}
```

    p = strstr(body, pat);
    if (!p) return false;
    p = strchr(p + strlen(pat), ':'); if (!p) return false;
    p = strchr(p, '"');               if (!p) return false;
    for (p++; *p && *p != '"' && i < sz-1; p++, i++) out[i] = *p;
    out[i] = '\0';
    return true;
}
static long json_int(const char* body, const char* field, long dflt)
{
    char pat[32]; const char* p;
    snprintf(pat, sizeof(pat), "\"%s\"", field);
    p = strstr(body, pat);
    if (!p) return dflt;
    p = strchr(p + strlen(pat), ':'); if (!p) return dflt;
    return strtol(p+1, NULL, 10);
}

static void handle_input(int fd, const char* body)
{
    char name[24]; char resp[64]; uint8_t code; long ms; int slot, n;
    if (!json_str(body, "key", name, sizeof(name))) {
        send_response(fd, 400, "text/plain", "missing \"key\"", 13);
        return;
    }
    code = key_for_name(name);
    if (code == 0) {
        static const char* valid = "valid keys: up down left right action "
            "pickup drop inv_left inv_right walk_run sleep stooge pause "
            "escape prisoner_1..4";
        send_response(fd, 400, "text/plain", valid, strlen(valid));
        return;
    }
    if (q_len >= INPUT_QUEUE_LEN) {
        send_response(fd, 400, "text/plain", "queue full", 10);
        return;
    }
    ms = json_int(body, "ms", 100);
    if (ms < 16) ms = 16;
    if (ms > 10000) ms = 10000;
    slot = (q_head + q_len) % INPUT_QUEUE_LEN;
    input_q[slot].code = code;
    input_q[slot].ticks = (int)(ms / 16);
    q_len++;
    n = snprintf(resp, sizeof(resp), "{\"queued\":%d}",
                 agent_input_queue_depth());
    send_response(fd, 202, "application/json", resp, n);
}

/* Called every tick from agent_api_tick(): advance the input machine. */
static void input_pump(void)
{
    static uint8_t active_code = 0;
    if (active_ticks > 0) {
        if (--active_ticks == 0) {           /* release, like glut_keyboard_up */
            key_down[active_code] = false;
            key_readonce[active_code] = false;
        }
        return;                              /* one key at a time */
    }
    if (q_len > 0) {
        active_code = input_q[q_head].code;
        active_ticks = input_q[q_head].ticks;
        q_head = (q_head + 1) % INPUT_QUEUE_LEN;
        q_len--;
        key_down[active_code] = true;        /* press, like glut_keyboard */
    }
}
